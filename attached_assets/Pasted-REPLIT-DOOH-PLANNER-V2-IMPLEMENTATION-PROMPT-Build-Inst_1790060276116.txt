// Module 1 State
const [filters, setFilters] = useState({
  market: 'UK',
  mode: 'Brand',
  brand: '& OTHER STORIES',
  category: 'Other',
  intelligenceTab: 'Audience',
  intelligenceFilters: {
    category: 'Consumer Behaviour',
    feature: 'Alcohol',
    value: 'Avg Spend On Beer Lager Ciders',
    logic: 'None'
  },
  topIndexing: '15%'
});

const [audienceDefinition, setAudienceDefinition] = useState({
  coverage: '2.3M people',
  details: ['12 postcodes in London', 'Budget tier only']
});

const [showSaveModal, setShowSaveModal] = useState(false);
const [audienceName, setAudienceName] = useState('');
const [audienceDescription, setAudienceDescription] = useState('');
const [savedAudiences, setSavedAudiences] = useState([]);
const [isSaving, setIsSaving] = useState(false);

// Validation
const isAudienceValid = () => audienceDefinition.coverage !== null;
const isDuplicateName = (name) => savedAudiences.some(a => a.name === name);

// Save Handler
const confirmSave = async () => {
  if (!audienceName || audienceName.length < 5) {
    showError('Name must be at least 5 characters');
    return;
  }

  if (isDuplicateName(audienceName)) {
    // Show warning but allow override
  }

  setIsSaving(true);
  try {
    const newAudience = {
      id: `AUD_${Date.now()}`,
      name: audienceName,
      description: audienceDescription,
      filters: filters,
      coverage: audienceDefinition.coverage,
      createdBy: 'Current User', // From Okta
      createdDate: new Date().toISOString(),
      isActive: true
    };

    // Mock backend save
    setSavedAudiences([...savedAudiences, newAudience]);
    
    // Show success toast
    showSuccess(`✓ "${audienceName}" saved successfully`);
    
    // Close modal
    setShowSaveModal(false);
    resetForm();
  } catch (error) {
    showError('Failed to save audience. Please try again.');
  } finally {
    setIsSaving(false);
  }
};
